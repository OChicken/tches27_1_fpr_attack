#+begin_src python
import numpy as np
import glob

OPT = "O0"
FALCON_N = 512
PREFIX = "."
DATA_PATH = f"{PREFIX}/{OPT}-falcon{FALCON_N}"
N_KEYS   = 100     # f_000 .. f_099  (and g_000 .. g_099)

def key_corr(fg, k, n_traces, traces_path):
    """One key -> one corr, via its own 3-trace reference (mean) and noise (std)."""
    traces = np.stack([
        np.load(f"{traces_path}/{fg}_{k:03}/trace_{t:02}.npy").astype(np.float64)
        for t in range(n_traces)
    ])                                     # (n_traces, SAMPLES)
    ref_trace = traces.mean(axis=0)        # this key's reference (not saved)
    noise     = traces.std(axis=0)         # this key's noise     (not saved; here for SNR)
    return float(np.mean([np.corrcoef(tr, ref_trace)[0, 1] for tr in traces]))
#+end_src

#+begin_src python
raw_path = f"{DATA_PATH}/traces_attack_raw"
aver_path = f"{DATA_PATH}/traces_attack"

def coaverage(fg):
    for k in range(N_KEYS):
        for n in range(10):
            print(3*n+0, 3*n+1, 3*n+2)
            t0 = np.load(f"{raw_path}/{fg}_{k:03}/trace_{3*n+0:02}.npy")
            t1 = np.load(f"{raw_path}/{fg}_{k:03}/trace_{3*n+1:02}.npy")
            t2 = np.load(f"{raw_path}/{fg}_{k:03}/trace_{3*n+2:02}.npy")
            traces = np.array([t0, t1, t2])
            trace  = traces.mean(axis=0)
            subprocess.run(["mkdir", "-p", f"{aver_path}/{fg}_{k:03}"])
            np.save(f"{aver_path}/{fg}_{k:03}/trace_{n:02}.npy", trace)
#+end_src
