#+begin_src python
# Parallelism is at the outer (per-key) level via ThreadPoolExecutor below, so
# pin torch/BLAS to 1 thread per worker here -- must be set before numpy/torch
# are imported (in Main), otherwise each MLP inference call would ALSO try to
# fan out across all cores and N_WORKERS threads would oversubscribe them.
for _v in ("OMP_NUM_THREADS", "MKL_NUM_THREADS", "OPENBLAS_NUM_THREADS",
           "NUMEXPR_NUM_THREADS", "VECLIB_MAXIMUM_THREADS"):
    os.environ.setdefault(_v, "1")

import sys, io, traceback
from concurrent.futures import ThreadPoolExecutor, as_completed
from utils import _ThreadRouter

# threads; torch/numpy release the GIL during heavy compute.
# Each worker is pinned to 1 BLAS/torch thread (see Common),
# so this is the actual degree of parallelism -- tune down
# if other processes are competing for the same cores.
N_WORKERS = int(os.environ.get("N_WORKERS", os.cpu_count()))
#+end_src

#+begin_src python
torch.set_num_threads(1)   # redundant with the env vars above, belt-and-suspenders
# Push config into the attack module's namespace -- the imported functions read
# OPT/PATH/MODEL/... from THERE, not from this session, so setting them here
# directly would have no effect.
attack.configure(opt=OPT, falcon_n=FALCON_N, model=MODEL, skip_shift55=SKIP_SHIFT55)

KEY_ID_LB = 0
KEY_ID_UB = 100

Falcon512_f_1000 = np.load("keys/Falcon512_f_1000.npy", allow_pickle=True)
Falcon512_g_1000 = np.load("keys/Falcon512_g_1000.npy", allow_pickle=True)
random.seed(2025)
ONLY_4_5 = False
try_random_num = 1000000
scaled_threshold = 0.99


def attack_one_key(key_id):
    # Own stage_stats (attack_key_repeat mutates in place) + own stdout buffer,
    # so workers never touch shared state.  The buffered text is returned and
    # printed later, strictly in key order.
    buf = io.StringIO()
    _router.set_buffer(buf)
    try:
        local_f = stats_init()
        local_g = stats_init()
        key_f = Falcon512_f_1000[key_id]
        key_g = Falcon512_g_1000[key_id]
        print(f"key {key_id:2}")
        print("\n#+begin_example")
        f_guess_dict, drop_re_f = attack_key_repeat(
            'f', key_f, key_id, local_f,
            MLE_N, scaled_threshold, only_4_5=ONLY_4_5)
        g_guess_dict, drop_re_g = attack_key_repeat(
            'g', key_g, key_id, local_g,
            MLE_N, scaled_threshold, only_4_5=ONLY_4_5)
        re, check_num = check_key_right(f_guess_dict, g_guess_dict, key_f, key_g,
                                        drop_re_f, drop_re_g, try_random_num)
        print(f"the {key_id}th key guess's state is {re}, #attempt = {check_num}")  # <
        print("#+end_example\n")
    except Exception:                # keep a crashed key's output for debugging
        traceback.print_exc(file=sys.stdout)
        re, check_num = -99, 0
        local_f, local_g = stats_init(), stats_init()
    finally:
        _router.clear_buffer()
    return key_id, re, check_num, local_f, local_g, buf.getvalue()


records = {}          # key_id -> (re, check_num, local_f, local_g)
pending = {}          # key_id -> captured text, waiting for its turn to print
next_print = KEY_ID_LB

_real_stdout = sys.stdout
sys.stdout = _router = _ThreadRouter(_real_stdout)   # install for the run
try:
    print(f"\n{datetime.now()}\n")
    with ThreadPoolExecutor(max_workers=N_WORKERS) as pool:
        futs = [pool.submit(attack_one_key, k)
                for k in range(KEY_ID_LB, KEY_ID_UB)]
        for fut in as_completed(futs):
            key_id, re, check_num, local_f, local_g, out = fut.result()
            records[key_id] = (re, check_num, local_f, local_g)
            pending[key_id] = out
            # flush the longest ready prefix so output is strictly by key_id;
            # a key that finished early waits in `pending` for its predecessors
            while next_print in pending:
                _real_stdout.write(pending.pop(next_print))
                _real_stdout.flush()
                next_print += 1
    print(f"\n{datetime.now()}\n")
finally:
    sys.stdout = _real_stdout        # always restore, even on error

# ---- merge single-threaded -> no locks needed ----
key_key_idx_list = []
error_list = []
check_num_list = []
succ_num_count = 0
total_stats = stats_init()
LKS = ("scaled", "mul_2_3", "mul_4_7") if SKIP_SHIFT55 else \
      ("scaled", "shift55", "mul_4_5", "mul_6_7")
for key_id in sorted(records):
    re, check_num, local_f, local_g = records[key_id]
    if re > 0:
        succ_num_count += 1
        key_key_idx_list.append(key_id)
        check_num_list.append(check_num)
    else:
        error_list.append(key_id)
    for lk in LKS:                                # fold this key into the totals
        for m in ("ok", "total", "misroute"):
            total_stats[lk][m] += local_f[lk][m] + local_g[lk][m]

print(f"the succ count is {succ_num_count}")
# cross-key per-stage correctness summary
report_sr_total('f + g', total_stats, skip_shift55=SKIP_SHIFT55)
