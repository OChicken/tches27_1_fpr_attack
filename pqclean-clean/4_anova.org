This document assesses whether the three targeted leakage sources --- =fpr_scaled=, =fpr_mul= shift55, and =fpr_mul= multiply --- carry statistically detectable leakage, and whether a classical Gaussian template attack can exploit it.

This document is used to
1. confirm leakage exists via pairwise Welch's t-test (TVLA);
2. visualise leakage strength with SOST/SNR diagrams and identify POIs;
3. evaluate classical Gaussian template attack success rates as a baseline.

The template attack success rates establish a baseline for what simple profiling can achieve.  Weaker sub-ranges (e.g. =fpr_mul= [16, 22]) motivate the ML-based discriminative classifier used in =5_profile.org=.

#+begin_src python
import glob

trace_downsampled = {}
trace_snr = {}
for key_id in LABEL_RANGE:
    trace_dir = f"{TRACES_PATH}/profile_{key_id}"
    files = sorted(glob.glob(f"{trace_dir}/trace_*.npy"))
    trace_raw = np.array([np.load(f, mmap_mode='r') for f in files])
    trace_lpf = trace_butter_lpf(trace_raw, ADC_FREQ, CUTOFF_FREQ)

    # Down-sampling
    n_trace = trace_lpf.shape[0]
    n_cycles = trace_lpf.shape[1] // PPC
    trace_downsampled[key_id] = trace_lpf[:, :n_cycles*PPC].reshape(n_trace, n_cycles, PPC).sum(axis=2)

    snr_raw = calc_trace_snr(trace_raw)
    snr_lpf = calc_trace_snr(trace_lpf)
    snr_downsampled = calc_trace_snr(trace_downsampled[key_id])
    trace_snr[key_id] = [snr_raw, snr_lpf, snr_downsampled]
#+end_src

#+begin_src python
print("#+begin_example")
for key_id in trace_snr.keys():

snr_vals = np.array(list(trace_snr.values()))

print(f"LPF: mean {np.mean(snr_vals[:,1]):7.4f} dB, min {np.min(snr_vals[:,1]):.4f} dB, max {np.max(snr_vals[:,1]):.4f} dB")
print(f"Down-sampled: mean {np.mean(snr_vals[:,2]):7.4f} dB, min {np.min(snr_vals[:,2]):.4f} dB, max {np.max(snr_vals[:,2]):.4f} dB")

So applying LPF can significantly improve the SNR. In the subsequent processing, we will implicitly apply LPF when loading the raw data.

Down-sampling step only applies to F-statistics, ICS, POI steps. For profiling (which uses POI), do NOT do down-sampling.

import numpy as np, pickle, glob
from scipy.signal import butter, filtfilt
from utils import class_coeffs_map_shift55

def lpf(x, fs, fc):
    b, a = butter(4, fc/(0.5*fs), btype='low')
    return filtfilt(b, a, x, axis=1)

# |value| → shift55 class
av2c = {}
for c, vs in class_coeffs_map_shift55.items():
    for v in vs:
        av2c[v] = c
#+end_src

#+begin_src python
out = {}
loc = pickle.load(open(f"{TRACES_PATH}/loc_shift55.pkl", "rb"))
elec = []; vals = {0: [], 1: []}
for key in range(-22, 23):
    cls = av2c.get(abs(key))        # only |value| in set0/set1 has a class
    if cls is None:
        continue
    files = sorted(glob.glob(f"{TRACES_PATH}/profile_{key}/trace_*.npy"))
    if len(files) < 2:  # >
        continue
    tl = lpf(np.array([np.load(f) for f in files]), ADC_FREQ, CUTOFF_FREQ)   # (2, nsamp)
    # cycle-0 (0x76) per coeff: sum the first PPC samples of each shift55 segment
    c0 = np.array([tl[:, loc[key][i][0]*PPC : loc[key][i][0]*PPC + PPC].sum(axis=1)
                   for i in range(512)])                        # (512, 2)
    elec.append(c0.var(axis=1, ddof=1))   # electronic var per coeff (var across 2 repeats)
    vals[cls].append(c0.ravel())          # all 512*2 cycle-0 values for this key's class

E = np.concatenate(elec).mean()                  # electronic floor
a0 = np.concatenate(vals[0]); a1 = np.concatenate(vals[1])
W = np.mean([a0.var(ddof=1), a1.var(ddof=1)])    # within-class total = E + algorithmic
S = np.var([a0.mean(), a1.mean()])               # between-class signal
A = max(W - E, 0.0)                               # algorithmic = total - electronic
out[OPT] = (E, S, A)
print(f"{OPT}: S/E={S/E:.3f}  A/E={A/E:.3f}  SNR=S/(E+A)={S/(E+A):.4f}")
