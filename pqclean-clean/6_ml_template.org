#+begin_src python
from TA_discriminative import classifier_train, classifier_inference, classifier_report
import torch
import torch.nn as nn
import torch.nn.functional as F

SEED = 42

class model_cnn(nn.Module):
    def __init__(self, n_class):
        super().__init__()
        k = 5
        p = k // 2  # "same" padding
        n_channel = 1
        # CNN
        self.conv1 = nn.Conv1d(n_channel, 4, kernel_size=k, padding=p)
        self.conv2 = nn.Conv1d(4, 4, kernel_size=k, padding=p)
        self.bn1 = nn.BatchNorm1d(4)
        self.bn2 = nn.BatchNorm1d(4)
        self.act = nn.Sigmoid()
        self.pool = nn.AvgPool1d(kernel_size=2, stride=2)
        # MLP
        self.fc1 = nn.LazyLinear(128)
        self.fc2 = nn.Linear(128, 128)
        self.fc3 = nn.Linear(128, n_class)
        self.dropout = nn.Dropout(0.1)
        self.relu = nn.ReLU()

    def forward(self, x):
        x = self.pool(self.act(self.bn1(self.conv1(x))))
        x = self.pool(self.act(self.bn2(self.conv2(x))))
        x = torch.flatten(x, 1)
        x = self.dropout(self.relu(self.fc1(x)))
        x = self.dropout(self.relu(self.fc2(x)))
        x = self.fc3(x)
        return x

class model_mlp(nn.Module):
    def __init__(self, n_class):
        super().__init__()
        self.flatten = nn.Flatten()
        self.bn1 = nn.BatchNorm1d(64)
        self.bn2 = nn.BatchNorm1d(64)
        self.bn3 = nn.BatchNorm1d(n_class)
        self.fc1 = nn.LazyLinear(64)
        self.fc2 = nn.Linear(64, 64)
        self.fc3 = nn.Linear(64, n_class)
        self.relu = nn.ReLU()

    def forward(self, x):
        x = self.flatten(x)
        x = F.selu(self.bn1(self.fc1(x)))
        x = F.selu(self.bn2(self.fc2(x)))
        x = F.selu(self.bn3(self.fc3(x)))
        x = F.relu(x)
        return x

def build_fpr_templates_ml(traces_dict, test_split, class_coeffs_map, model_name, num_epochs=50):
    np.random.seed(SEED)
    torch.manual_seed(SEED)

    traces_train, traces_test, label_train, label_test = \
        select_and_split(traces_dict, class_coeffs_map, test_split)
    n_class = len(class_coeffs_map.keys())
    model = model_mlp(n_class)
    classifier_train(traces_train, label_train, model, model_name, num_epochs)
    # we actually don't need y_prob to attack; in case you need it to check bug, I also export it
    y_prob = classifier_inference(traces_test, model_name)
    classifier_report(traces_test, label_test, model_name, verbose=True)

** Observation

CNN gives no meaningful improvement over MLP: the per-stage CNN−MLP accuracy gap is within ±2 pp, mixed in sign, averages ≈0 (slightly negative on Os), and CNN does not lift the shift55 bottleneck (Os 0.9208 → 0.9215). So it is sufficient to read the MLP results only; the rest of this observation does so.

The accuracy, balanced-accuracy and macro-f1 tables are identical to 3-4 decimals at every stage, so the three carry the same information and no stage hides a class collapse; read the Accuracy table and ignore the other two. scaled and mul [4,5] saturate at ≈1.0000 for both opts and both models, so they separate nothing between O3 and Os.

The discriminating stage is shift55: it is the *only* stage where O3 beats Os (MLP 0.9596 v.s. 0.9208, CNN 0.9661 v.s. 0.9215), and the loss table agrees (O3's shift55 loss is lower, 0.1128 v.s. 0.2000 for MLP). On every higher-magnitude mul stage the ranking *flips* --- Os is the stronger profiler (MLP mul [8,11] 0.8867 v.s. 0.9580, mul [12,15] 0.9648 v.s. 0.9834, mul [16,22] 0.7623 v.s. 0.8588), and CNN shows the same pattern.

So the profiling numbers alone would suggest Os is the better target, which is the trap: Os's high scaled/mul accuracy is an illusion. Its one real weakness --- shift55 --- is exactly the stage the attack cannot route around, and the shift55 ANOVA R2/SNR, not the MLP/CNN accuracy, is what predicts Os failing at the attack stage.
