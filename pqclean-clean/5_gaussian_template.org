#+begin_src python
import glob

trace_lpf = {}
trace_snr = {}
for key_id in LABEL_RANGE:
    trace_dir = f"{TRACES_PATH}/profile_{key_id}"
    files = sorted(glob.glob(f"{trace_dir}/trace_*.npy"))
    trace_raw = np.array([np.load(f) for f in files])
    trace_lpf[key_id] = trace_butter_lpf(trace_raw, ADC_FREQ, CUTOFF_FREQ)

    snr_raw = calc_trace_snr(trace_raw)
    snr_lpf = calc_trace_snr(trace_lpf[key_id])
    trace_snr[key_id] = [snr_raw, snr_lpf]
#+end_src

#+begin_src python
print("#+begin_example")
for key_id in trace_snr.keys():
    snr_raw, snr_lpf = trace_snr[key_id]
    print(f"  key_id={key_id:3}: {snr_raw:.4f} dB → {snr_lpf:.4f} dB")
snr_vals = np.array(list(trace_snr.values()))
print(f"Raw: mean {np.mean(snr_vals[:,0]):7.4f} dB, min {np.min(snr_vals[:,0]):.4f} dB, max {np.max(snr_vals[:,0]):.4f} dB")
print(f"LPF: mean {np.mean(snr_vals[:,1]):7.4f} dB, min {np.min(snr_vals[:,1]):.4f} dB, max {np.max(snr_vals[:,1]):.4f} dB")
print("#+end_example\n")
#+end_src

So applying LPF can significantly improve the SNR. In the subsequent processing, we will implicitly apply LPF when loading the raw data.

#+begin_src python
import pickle

with open(f"{TRACES_PATH}/loc_scaled.pkl", 'rb') as f:
    loc_scaled = pickle.load(f)
with open(f"{TRACES_PATH}/loc_mul.pkl", 'rb') as f:
    loc_mul = pickle.load(f)
with open(f"{TRACES_PATH}/loc_shift55.pkl", 'rb') as f:
    loc_shift55 = pickle.load(f)
with open(f"{TRACES_PATH}/loc_mul_shift55.pkl", 'rb') as f:
    loc_mul_shift55 = pickle.load(f)
#+end_src

#+begin_src python
trace_scaled = {}
trace_mul = {}
trace_shift55 = {}
trace_mul_shift55 = {}

for key_id in LABEL_RANGE:
    trace_scaled[key_id] = np.array([
        trace_lpf[key_id][:, loc_scaled[key_id][i][0]*PPC : loc_scaled[key_id][i][1]*PPC]
        for i in range(FALCON_N)
    ]).transpose(1, 0, 2).reshape(-1, CYC_SCALED*PPC)
    trace_mul[key_id] = np.array([
        trace_lpf[key_id][:, loc_mul[key_id][i][0]*PPC : loc_mul[key_id][i][1]*PPC]
        for i in range(FALCON_N)
    ]).transpose(1, 0, 2).reshape(-1, CYC_MUL*PPC)
    trace_shift55[key_id] = np.array([
        trace_lpf[key_id][:, loc_shift55[key_id][i][0]*PPC : loc_shift55[key_id][i][1]*PPC]
        for i in range(FALCON_N)
    ]).transpose(1, 0, 2).reshape(-1, CYC_SHIFT55*PPC)
    trace_mul_shift55[key_id] = np.array([
        trace_lpf[key_id][:, loc_mul_shift55[key_id][i][0]*PPC : loc_mul_shift55[key_id][i][1]*PPC]
        for i in range(FALCON_N)
    ]).transpose(1, 0, 2).reshape(-1, CYC_MUL_SHIFT55*PPC)

print("#+begin_example")
print(f"shape summary: for FALCON_N = {FALCON_N}, PPC = {PPC},")
print(f"  trace_scaled[coeff]:      {trace_scaled[0].shape}")
print(f"  trace_mul[coeff]:         {trace_mul[0].shape}")
print(f"  trace_shift55[coeff]:     {trace_shift55[0].shape}")
print(f"  trace_mul_shift55[coeff]: {trace_mul_shift55[0].shape}")
print("recall:")
print(f"  CYC_SCALED      = {CYC_SCALED}")
print(f"  CYC_MUL         = {CYC_MUL}")
print(f"  CYC_SHIFT55     = {CYC_SHIFT55}")
print(f"  CYC_MUL_SHIFT55 = {CYC_MUL_SHIFT55}")
print("#+end_example")
#+end_src
